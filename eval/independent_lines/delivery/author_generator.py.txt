        if len(offs) == 1:
            lines[-1]["off"] = span if lines[-1]["off"] != span else 0
    add("clean", "CLEAN", "standard", span, lines,
        "Ordinary claim: every unit count, date and code pairing is within the stated rules.")

# ================================================================ build
PROVIDERS = []
for k in range(1, 25):
    PROVIDERS.append({"provider_id": f"PRV{k:03d}", "specialty": f"S{rng.randint(1, 6)}",
                      "region": f"R{rng.randint(1, 4)}", "group_id": f"G{rng.randint(1, 8)}"})

order = list(range(len(cases)))
rng.shuffle(order)
assert len(cases) == 108

claims, labels = [], []
member_seq = 0
members = []  # (member dict, provider_id used)
for pos, ci in enumerate(order, start=1):
    c = cases[ci]
    cid = f"CLM{pos:04d}"
    prov = rng.choice(PROVIDERS)
    reuse = c["type"] == "clean" and members and rng.random() < 0.25
    if reuse:
        mem, used = rng.choice(members)
        while prov["provider_id"] == used:
            prov = rng.choice(PROVIDERS)
    else:
        member_seq += 1
        mem = {"member_id": f"MEM{member_seq:03d}",
               "age_band": rng.choice(["0-17", "18-29", "30-39", "40-49", "50-64", "65+"]),
               "sex": rng.choice(["F", "M"]), "plan_type": rng.choice(["PPO", "HMO", "EPO", "POS"]),
               "region": f"R{rng.randint(1, 4)}"}
        members.append((mem, prov["provider_id"]))
    if c["start"]:
        frm = date.fromisoformat(c["start"])
    else:
        frm = date(2026, 11, 1) + timedelta(days=rng.randint(0, 270))
    to = frm + timedelta(days=c["span"])
    dxs = rng.sample(DX, rng.choice([1, 2, 3]))
    lines, tot = [], Decimal("0.00")
    refs_t, refs_w = [], []
    for k, l in enumerate(c["lines"], start=1):
        amt = (BASE[l["code"]] * l["units"]).quantize(Decimal("0.01"))
        tot += amt
        lines.append({"line_no": k, "code": l["code"], "units": l["units"],
                      "billed_amount": f"{amt:.2f}",
                      "service_date": (frm + timedelta(days=l["off"])).isoformat(),
                      "modifiers": l["mods"], "dx_pointer": rng.randint(1, len(dxs))})
        if l["tag"] == "T":
            refs_t.append({"claim_id": cid, "line_no": k})
        elif l["tag"] == "W":
            refs_w.append({"claim_id": cid, "line_no": k})
    claims.append({"claim_id": cid, "member": mem, "provider": prov,
                   "service_from": frm.isoformat(), "service_to": to.isoformat(),
                   "place_of_service": rng.choice(["PS1", "PS2", "PS3"]),
                   "total_billed": f"{tot:.2f}", "diagnoses": dxs, "lines": lines})
    def fmt(refs):
        ns = [r["line_no"] for r in refs]
        if not ns:
            return ""
        return ("line " + str(ns[0])) if len(ns) == 1 else ("lines " + " and ".join(map(str, ns)))
    reason = c["reason"].replace("{T}", fmt(refs_t).capitalize()).replace("{W}", fmt(refs_w).capitalize())
    reason = reason.replace("Lines ", "Lines ").replace(": Line", ": line")
    # lowercase "Line N" mid-sentence references after a semicolon
    reason = reason.replace("; Line", "; line").replace("; Lines", "; lines")
    if c["label"] == "TRUE" and c["type"] != "clean":
        assert refs_t, cid
    if c["label"] != "TRUE":
        assert not refs_t
    labels.append({"case_id": f"L{pos:03d}", "type": c["type"], "label": c["label"],
                   "claim_ids": [cid], "provider_id": None, "reason": reason,
                   "case_kind": c["kind"], "line_refs": refs_t if c["label"] == "TRUE" else [],
                   "tempting_line_refs": refs_w})

os.makedirs(OUT, exist_ok=True)
with open(f"{OUT}/claims.jsonl", "w") as f:
    for cl in claims:
        f.write(json.dumps(cl) + "\n")
with open(f"{OUT}/labels.jsonl", "w") as f:
    for lb in labels:
        f.write(json.dumps(lb) + "\n")
with open(f"{OUT}/rules.json", "w") as f:
    json.dump(RULES, f, indent=2)
    f.write("\n")
print("seed", SEED, "claims", len(claims), "labels", len(labels))
